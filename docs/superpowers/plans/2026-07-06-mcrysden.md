# mcrysden Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `mcrysden`, a native macOS (Metal) crystal-and-molecule viewer — a from-scratch port of XCrySDen's v1 core scope: read XSF/AXSF/XYZ/PDB, render ball-stick / space-fill / wireframe / polyhedral + 2D modes, supercell + slab, save/load state, headless PNG export. Launched from terminal; no `.app` bundle.

**Architecture:** Swift Package Manager package. `@main NSApplication` (`MolVisApp`) dispatches between GUI (window = NSSplitView of SwiftUI sidebar + MTKView Metal canvas) and headless (`--export`). A small C library `MolEnvParse` (ported from XCrySDen `readstrf.c` / `readXYZ.c` / `pdb.c`) owns file parsing; Swift copies out of its heap into value-type `Scene`. `Renderer.encode(to:target:viewport:scene:camera:)` draws a `Scene` to an arbitrary `MTKTexture` so the same draw path serves both on-screen (`MTKView` drawable) and offscreen (a transient texture for PNG export).

**Tech Stack:** Swift 5.9, Metal 3, AppKit, SwiftUI (sidebar), SPM, XCTest. macOS 14+ / Xcode 15+. C parser compiled as an SPM `CLibrary` / static-lib target.

## Global Constraints

- macOS 14 (Sonoma) minimum deployment target; build with Swift 5.9 / Xcode 15+.
- Apple Silicon primary; Intel also builds (no arch-specific code outside Metal).
- No third-party Swift packages. C parser uses only libc (`stdlib.h stdio.h string.h math.h`).
- Metal shader API: Metal 3, no MetalFX, no ray tracing. One render pipeline per geometry class; instancing for atoms/bonds; lines for cell frame + axes.
- C parser must return a heap `MolEnvScene*`; Swift copies then calls `molenv_scene_free`. No C pointer lives across an async boundary.
- `Renderer.encode(to:target:viewport:scene:camera:)` takes the target texture as a parameter — never assumes an `MTKView`. This is the design invariant that makes headless export the same code as on-screen draw.
- Executable name: `mcrysden`; state files: `.mvis-state` (JSON); repo root: `/Users/mao/dev/mcrysden`.
- Every task commits at the end; every task has a green-test gate before the commit.

- [ ] **Step 5: Author Parser.swift**

```swift
import Foundation

enum ParseError: Error, CustomStringConvertible {
    case parse(path: String, line: Int, reason: String)
    case io(path: String, reason: String)
    var description: String {
        switch self {
        case .parse(let p, let line, let r): return "\(p):\(line): \(r)"
        case .io(let p, let r): return "\(p): \(r)"
        }
    }
}

struct Atom { var coord: SIMD3<Float>; var atomicNumber: Int; var label: String }
struct Bond  { var i: Int; var j: Int }
struct Cell  { var a: SIMD3<Float>; var b: SIMD3<Float>; var c: SIMD3<Float> }

struct LoadedScene {
    var atoms: [Atom] = []
    var bonds: [Bond] = []
    var cell: Cell?
    var isCrystal: Bool = false
    var periodicDim: Int = 3
    var title: String = ""
}

enum Parser {
    static func load(_ url: URL) throws -> LoadedScene {
        guard FileManager.default.fileExists(atPath: url.path) else {
            throw ParseError.io(path: url.path, reason: "file not found")
        }
        let cPath = url.path.cString(using: .utf8)!
        let ext = url.pathExtension.lowercased()
        let scene: OpaquePointer?
        switch ext {
        case "xsf": scene = OpaquePointer(parse_xsf(cPath))
        case "xyz": scene = OpaquePointer(parse_xyz(cPath))
        case "pdb": scene = OpaquePointer(parse_pdb(cPath))
        case "axsf": scene = OpaquePointer(parse_axsf(cPath, 0))
        default: throw ParseError.io(path: url.path, reason: "unknown extension \(ext)")
        }
        guard let scene else {
            let msg = String(cString: molenv_last_error())
            // "path:line: reason" or "path: reason"
            let parts = msg.components(separatedBy: ": ")
            if parts.count >= 3, let ln = Int(parts[1]) {
                throw ParseError.parse(path: String(parts[0]), line: ln, reason: parts.dropFirst(2).joined(separator: ": "))
            }
            throw ParseError.parse(path: url.path, line: 0, reason: msg)
        }
        defer { molenv_scene_free(scene) }
        let s = scene.assumingMemoryBound(to: MolEnvScene.self).pointee
        var out = LoadedScene()
        out.isCrystal = s.is_crystal != 0
        out.periodicDim = s.periodic_dim
        out.title = String(cString: s.title)
        if s.natoms > 0 && s.atoms != nil {
            let buf = UnsafeBufferPointer(start: s.atoms, count: s.natoms)
            out.atoms = buf.map { Atom(coord: SIMD3($0.coord[0], $0.coord[1], $0.coord[2]),
                                      atomicNumber: $0.atomic_number,
                                      label: String(cString: $0.label)) }
        }
        if s.nbonds > 0 && s.bonds != nil {
            let buf = UnsafeBufferPointer(start: s.bonds, count: s.nbonds)
            out.bonds = buf.map { Bond(i: $0.i, j: $0.j) }
        }
        if s.is_crystal != 0 {
            out.cell = Cell(a: SIMD3(s.cell[0][0], s.cell[0][1], s.cell[0][2]),
                           b: SIMD3(s.cell[1][0], s.cell[1][1], s.cell[1][2]),
                           c: SIMD3(s.cell[2][0], s.cell[2][1], s.cell[2][2]))
        }
        return out
    }
}
```

#include <stdio.h>
#include <ctype.h>

static int atomic_number_from_label(const char *lbl) {
    // crude symbol→Z; real version in Task 5 will ship table from C
    typedef struct { const char* sym; int z; } E;
    static const E table[] = {
        {"H",1},{"He",2},{"Li",3},{"Be",4},{"B",5},{"C",6},{"N",7},{"O",8},{"F",9},{"Ne",10},
        {"Na",11},{"Mg",12},{"Al",13},{"Si",14},{"P",15},{"S",16},{"Cl",17},{"Ar",18},
        {"K",19},{"Ca",20},{"Ti",22},{"V",23},{"Cr",24},{"Mn",25},{"Fe",26},{"Co",27},
        {"Ni",28},{"Cu",29},{"Zn",30},{"Ga",31},{"Ge",32},{"As",33},{"Se",34},{"Br",35},
        {"Kr",36},{"Zr",40},{"Nb",41},{"Mo",42},{"Tc",43},{"Ru",44},{"Rh",45},{"Pd",46},
        {"Ag",47},{"Cd",48},{"In",49},{"Sn",50},{"Sb",51},{"Te",52},{"I",53},{"Xe",54},
        {"Pt",78},{"Au",79},{"Hg",80},{"Tl",81},{"Pb",82},{"Bi",83}
    };
    char s[4] = {0};
    for (int i = 0; i < 3 && lbl[i]; i++) s[i] = lbl[i];
    for (size_t i = 0; i < sizeof(table)/sizeof(table[0]); i++)
        if (strcmp(s, table[i].sym) == 0) return table[i].z;
    return 0;
}

static MolEnvScene* parse_xyz_impl(const char *path) {
    FILE *fp = fopen(path, "r");
    if (!fp) { set_error(path, 0, "cannot open file"); return NULL; }
    MolEnvScene *s = new_scene(path);
    if (!s) { fclose(fp); return NULL; }
    int natoms = 0;
    if (fscanf(fp, "%d", &natoms) != 1 || natoms <= 0) {
        set_error(path, 1, "could not parse atom count"); goto fail;
    }
    char line[1024];
    if (!fgets(line, sizeof(line), fp)) goto fail; // consume rest of count line
    if (!fgets(line, sizeof(line), fp)) goto fail; // comment line
    snprintf(s->title, sizeof(s->title), "%s", line);
    // trim trailing newline
    size_t tlen = strlen(s->title);
    while (tlen && (s->title[tlen-1]=='\n'||s->title[tlen-1]=='\r')) s->title[--tlen] = '\0';
    s->atoms = calloc(natoms, sizeof(MolEnvAtom));
    if (!s->atoms) { set_error(path, 0, "out of memory"); goto fail; }
    for (int i = 0; i < natoms; i++) {
        char sym[16] = {0};
        double x = 0, y = 0, z = 0;
        if (!fgets(line, sizeof(line), fp)) { set_error(path, i+3, "unexpected EOF"); goto fail; }
        if (sscanf(line, "%15s %lf %lf %lf", sym, &x, &y, &z) != 4) {
            set_error(path, i+3, "malformed atom line"); goto fail;
        }
        MolEnvAtom *a = &s->atoms[s->natoms++];
        a->coord[0] = (float)x; a->coord[1] = (float)y; a->coord[2] = (float)z;
        a->atomic_number = atomic_number_from_label(sym);
        snprintf(a->label, sizeof(a->label), "%s", sym);
    }
    // auto-bond after parse
    s->bonds = make_bonds(s, 1.3, &s->nbonds);
    fclose(fp);
    return s;
fail:
    fclose(fp);
    molenv_scene_free(s);
    return NULL;
}

import Foundation
import simd

extension Scene {
    init(loaded: LoadedScene, displayMode: DisplayMode = .ballStick) {
        self.atoms = loaded.atoms
        self.bonds = loaded.bonds
        self.cell = loaded.cell
        self.title = loaded.title
        self.displayMode = displayMode
        self.isCrystal = loaded.isCrystal
        self.periodicDim = loaded.periodicDim
    }

    /// In Angstroms; supports fractional cell via cell vectors when isCrystal.
    var isCrystal: Bool { get { cell != nil } set { if !newValue { cell = nil } } }
    var periodicDim: Int = 3

    var centroid: SIMD3<Float> {
        guard !atoms.isEmpty else { return SIMD3(0,0,0) }
        var sum = SIMD3<Float>.zero
        for a in atoms { sum += a.coord }
        return sum / Float(atoms.count)
    }

    func boundingSphere() -> (center: SIMD3<Float>, radius: Float) {
        let c = centroid
        var r: Float = 0
        for a in atoms { r = max(r, distance(a.coord, c)) }
        return (c, r)
    }
}

- [ ] **Step 4: Bond heuristic — in molenv_parse.c add `molenv_covalent_radius(int z)` and `make_bonds`**

```c
static float molenv_covrad[119];

static void init_covrad(void) {
    for (int i = 0; i < 119; i++) molenv_covrad[i] = 1.5f;
    molenv_covrad[1]=0.31f; molenv_covrad[6]=0.76f; molenv_covrad[7]=0.71f;
    molenv_covrad[8]=0.66f; molenv_covrad[14]=1.11f; molenv_covrad[15]=1.07f;
    molenv_covrad[16]=1.05f; molenv_covrad[17]=1.02f;
}

static float dist3(const float *a, const float *b) {
    float d0=a[0]-b[0], d1=a[1]-b[1], d2=a[2]-b[2];
    return sqrtf(d0*d0+d1*d1+d2*d2);
}

static MolEnvBond* make_bonds(const MolEnvScene *s, float factor, int *out_nbonds) {
    static int ready = 0;
    if (!ready) { init_covrad(); ready = 1; }
    int cap = s->natoms * 4, nb = 0;
    MolEnvBond *b = calloc(cap, sizeof(MolEnvBond));
    for (int i = 0; i < s->natoms; i++) {
        for (int j = i+1; j < s->natoms; j++) {
            float r = molenv_covrad[s->atoms[i].atomic_number] + molenv_covrad[s->atoms[j].atomic_number];
            if (dist3(s->atoms[i].coord, s->atoms[j].coord) <= r * factor) {
                if (nb >= cap) { cap *= 2; b = realloc(b, cap*sizeof(MolEnvBond)); }
                b[nb].i = i; b[nb].j = j; nb++;
            }
        }
    }
    *out_nbonds = nb;
    return b;
}
```

using namespace metal;

struct VertexIn { float3 position [[attribute(0)]]; float3 normal [[attribute(1)]]; };
struct InstanceData { float4x4 model; float3 color; float radius; };
struct FrameData { float4x4 view; float4x4 proj; float3 lightDir; };

struct VInOut {
    float4 position [[position]];
    float3 normal;
    float3 worldPos;
    float3 color;
};

vertex VInOut v_main(VertexIn in [[stage_in]],
                     constant InstanceData &inst [[buffer(1)]],
                     constant FrameData &f [[buffer(2)]],
                     uint iid [[instance_id]]) {
    VInOut o;
    float3 p = in.position * inst.radius + (float3)(inst.model[3].xyz); // model translation column + scale
    o.worldPos = p;
    o.normal = in.normal;
    o.color = inst.color;
    o.position = f.proj * f.view * float4(p, 1.0);
    return o;
}

fragment float4 f_main(VInOut in [[stage_in]], constant FrameData &f [[buffer(2)]]) {
    float3 N = normalize(in.normal);
    float3 L = normalize(f.lightDir);
    float diff = max(dot(N, L), 0.0);
    float3 ambient = in.color * 0.35;
    float3 diffuse = in.color * diff * 0.65;
    return float4(ambient + diffuse, 1.0);
}

// Line shader for cell frame + axes

struct LineVOut { float4 position [[position]]; float3 color; };
vertex LineVOut lv_main(LineVertexIn in [[stage_in]], constant FrameData &f [[buffer(2)]],
                        constant float3 &color [[buffer(3)]]) {
    LineVOut o; o.color = color; o.position = f.proj * f.view * float4(in.position, 1.0); return o;
}

import Foundation

enum StateStore {
    static func save(_ scene: Scene, camera: Camera?, to url: URL) throws {
        var payload: [String: Any] = ["version": 1]
        let enc = JSONEncoder(); enc.outputFormatting = [.prettyPrinted, .sortedKeys]
        payload["scene"] = try JSONSerialization.jsonObject(with: try enc.encode(scene))
        if let camera { payload["camera"] = try JSONSerialization.jsonObject(with: try enc.encode(camera)) }
        let data = try JSONSerialization.data(withJSONObject: payload, options: [.prettyPrinted])
        try data.write(to: url)
    }
    static func load(_ scene: inout Scene, camera: inout Camera?, from url: URL) throws {
        let data = try Data(contentsOf: url)
        let obj = try JSONSerialization.jsonObject(with: data) as? [String: Any]
        guard let obj else { throw ParseError.io(path: url.path, reason: "bad state file") }
        if let v = obj["version"] as? Int, v > 1 { throw ParseError.parse(path: url.path, line: 0, reason: "state version \(v) too new") }
        let dec = JSONDecoder()
        if let s = obj["scene"] {
            scene = try dec.decode(Scene.self, from: try JSONSerialization.data(withJSONObject: s))
        }
        if let c = obj["camera"] { camera = try dec.decode(Camera.self, from: try JSONSerialization.data(withJSONObject: c)) }
    }
}

```swift
import AppKit
import Metal
enum PngExporter {
    static func export(scene: Scene, camera: Camera?, to url: URL, size: CGSize) throws {
        guard let device = MTLCreateSystemDefaultDevice() else { throw "no GPU" }
        let renderer = try Renderer(device: device)
        renderer.scene = scene
        let desc = MTLTextureDescriptor.texture2DPixelFormat(.rgba8Unorm, width: Int(size.width), height: Int(size.height), mipmapped: false)
        desc.usage = [.renderTarget, .shaderRead]
        guard let tex = device.makeTexture(descriptor: desc) else { throw "no tex" }
        let q = device.makeCommandQueue()!
        let cb = q.makeCommandBuffer()!
        var cam = camera ?? {
            var c = Camera()
            let (cen, r) = scene.boundingSphere()
            c.center = cen; c.distance = max(8, r*3)
            return c
        }()
        let viewport = MTLViewport(originX: 0, originY: 0, width: Double(size.width), height: Double(size.height), znear: 0, zfar: 1)
        renderer.encode(to: cb, target: tex, viewport: viewport, camera: cam)
        let blit = cb.makeBlitCommandEncoder()!
        blit.synchronize(resource: tex)
        blit.endEncoding()
        cb.commit(); cb.waitUntilCompleted()
        // tex → CGImage → PNG
        let bytesPerRow = Int(size.width)*4
        var bytes = [UInt8](repeating: 0, count: bytesPerRow*Int(size.height))
        tex.getBytes(&bytes, bytesPerRow: bytesPerRow, from: MTLRegionMake2D(0,0,Int(size.width),Int(size.height)), mipmapLevel: 0)
        let ctx = CGContext(data: &bytes, width: Int(size.width), height: Int(size.height), bitsPerComponent: 8, bytesPerRow: bytesPerRow, space: CGColorSpaceCreateDeviceRGB(), bitmapInfo: CGImageAlphaInfo.premultipliedLast.rawValue | CGBitmapInfo.byteOrder32Big.rawValue)!
        let cg = ctx.makeImage()!
        let rep = NSBitmapImageRep(cgImage: cg)
        let png = rep.representation(using: .png, properties: [:])!
        try png.write(to: url)
    }
}
```
(For headless pipeline to compile in `--export` path, `Renderer` must not require a window.)
